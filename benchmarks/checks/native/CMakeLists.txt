cmake_minimum_required(VERSION 3.21)
project(LassiNativeFixtures LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
add_executable(reference "${LASSI_REFERENCE}")
set_target_properties(reference PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LASSI_OUTPUT_DIRECTORY}")
target_compile_options(reference PRIVATE -O2 -ffp-contract=off)
if(LASSI_KIND STREQUAL "mutual-information")
  find_package(ITK REQUIRED COMPONENTS ITKCommon ITKStatistics ITKImageCompose)
  include(${ITK_USE_FILE})
  target_sources(reference PRIVATE mutual-information-bridge.cpp)
  target_link_libraries(reference PRIVATE ${ITK_LIBRARIES})
elseif(LASSI_KIND STREQUAL "randomized-svd")
  find_package(RandLAPACK CONFIG REQUIRED)
  target_sources(reference PRIVATE randomized-svd-bridge.cpp)
  target_link_libraries(reference PRIVATE RandLAPACK::RandLAPACK)
elseif(LASSI_KIND STREQUAL "cfd")
  target_sources(reference PRIVATE cfd-bridge.cpp)
  target_include_directories(reference PRIVATE "${LASSI_ROOT}/benchmarks/rodinia/openmp/cfd" "${LASSI_OMP_INCLUDE}")
else()
  message(FATAL_ERROR "Unknown native C++ fixture ${LASSI_KIND}")
endif()
if(LASSI_SMALL_TEST)
  target_compile_definitions(reference PRIVATE LASSI_CFD_N=384 LASSI_MI_SIDE=32 LASSI_RSVD_M=64 LASSI_RSVD_N=32 LASSI_RSVD_K=8)
endif()
